"""One-way audit projection of live native records; never restores authority."""
from collections.abc import Mapping
from dataclasses import fields, is_dataclass
from hedgehog.kernel import abi_v01 as abi

def plain(value):
    # The ABI owns receipt/artifact validation and its immutable JSON projection.
    if isinstance(value, abi.KernelArtifactV01):
        return abi.kernel_artifact_to_plain_dict_v01(value)
    if is_dataclass(value):
        return {field.name: plain(getattr(value, field.name)) for field in fields(value)}
    if isinstance(value, Mapping):
        if not all(type(key) is str for key in value):
            raise TypeError('audit_mapping_key')
        return {key: plain(item) for key, item in value.items()}
    if isinstance(value, (tuple, list)):
        return [plain(item) for item in value]
    if value is None or type(value) in (str, int, bool, float):
        return value
    raise TypeError('audit_unsupported:' + type(value).__name__)
