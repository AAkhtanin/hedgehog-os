"""Precreated peer entrypoint. Import performs no Work, Root, transport or effects."""
import json
import pathlib
from hedgehog.external_drs import gate5_contracts_v01 as c, gate5_native_v01 as n, gate5_exchange_v01 as e
from hedgehog.external_drs.gate5_exchange_v01 import PipeChannel
from . import domain as d, policy as p, effect, source, transport

def new_counts():
    return dict(source_work=0,requester_work=0,search=0,transport_requests=0,
        root_reviews=0,dispatch=0,mutations=0,receipt_reuse=0,pure_verification=0,
        runtime_semantic_calls=0,provider_calls=0)

def difference(after,before):
    return {k:after[k]-before[k] for k in after}

def report_failure(request, exc, role, folder, counts, historical=False):
    status = exc.status
    if counts['dispatch'] or counts['mutations']:
        status = 'CURRENT_STATUS_UNKNOWN'
    refs = {}
    evidence = dict(reason=str(exc),phase='EVENT_REFUSAL',
                    status=status,classification='OBSERVED_LOCAL_REFUSAL')
    e.immutable(folder/'refusal.json',evidence)
    if not historical:
        review = n.root_review(p.ROOTS[role],'transaction:football:refusal',
            'candidate:football:refusal:'+c.sha([request,str(exc)]),
            'request:football:'+c.sha(request),dict(required_predicate_satisfied=False),
            dict(request=request,reason=str(exc),phase='EVENT_REFUSAL'),e.clock())
        counts['root_reviews'] += 1
        n.save(folder/'refusal_review.json',n.root_plain(review))
        refs['refusal_review'] = dict(file='refusal_review.json',id=review[2].decision_id)
    refs['refusal'] = dict(file='refusal.json',id='refusal:'+c.sha(evidence))
    if (folder/'receipt.json').exists():
        receipt = json.loads((folder/'receipt.json').read_text())
        refs['receipt_history'] = dict(file='receipt.json',id=receipt['artifact_id'])
    return d.report(request,status,search=counts['search'],dispatch=counts['dispatch'],
                    mutations=counts['mutations']),refs

def history_requester(request,state_root,folder,counts):
    path = state_root/'accepted'/(effect.scope_key(request)+'.json')
    d.need(path.exists(),'NO_SAVED_RECEIPT','CURRENT_STATUS_UNKNOWN')
    saved = json.loads(path.read_text())
    d.need(saved['body']['booking'] is not None,'NO_SAVED_BOOKING','CURRENT_STATUS_UNKNOWN')
    d.need(d.constraints(saved['request']) == d.constraints(request),'SAVED_REQUEST_CHANGED')
    if request['operation'] == 'CONFIRM_MOCK_BOOKING':
        d.need(saved['request'] == request,'SAME_KEY_CHANGED_PAYLOAD')
        counts['receipt_reuse'] += 1
    else:
        counts['pure_verification'] += 1
    booking = saved['body']['booking']
    proof = saved['authenticated']
    mh = proof['bundle']['manifest']['manifest_id'].split(':')[1]
    # Verify immutable old signatures against the separately saved local trust
    # input. No temporal validation, new Root acceptance, or current grant follows.
    try:
        pointer = c.verify(proof['descriptor'],proof['peer_keys'],'POINTER',mh)
        c.verify(proof['status'],proof['peer_keys'],'STATUS',mh)
        release = c.verify(proof['bundle']['release'],proof['peer_keys'],'RELEASE',mh)
        d.need(pointer['body_sha256'] == c.sha(saved['body']) and
               proof['bundle']['body'] == saved['body'] and
               release['body_hash'] == c.sha(saved['body']), 'SAVED_SIGNATURE_BINDING')
    except ValueError as exc:
        raise d.Refusal(str(exc),'CURRENT_STATUS_UNKNOWN') from exc
    n.save(folder/'signed_history.json',proof)
    d.need(saved['result']['receipt_ref'] == booking['receipt_ref'] and
           saved['result']['slots'] == booking['slots'],'SAVED_RECEIPT_BINDING',
           'CURRENT_STATUS_UNKNOWN')
    # No foreign Host, Root or grant is reconstructed from these history bytes.
    n.save(folder/'offer_body.json',saved['body'])
    n.save(folder/'requester_work.json',saved['work'])
    n.save(folder/'requester_review.json',saved['review'])
    e.immutable(folder/'history.json',dict(classification='SAVED_SIGNED_SOURCE_EVIDENCE_HISTORY',
        original_request=saved['request'],original_result=saved['result'],
        receipt_ref=booking['receipt_ref'],new_source_fetch=False,new_root_decision=False))
    result = d.report(request,'VERIFIED' if request['operation'] == 'VERIFY_EXISTING'
                      else 'MOCK_BOOKED',booking['slots'],saved['result']['total_minor'],
                      saved['result']['source'],booking['receipt_ref'])
    refs = dict(history=dict(file='history.json',id=booking['receipt_ref']),
        prior_requester_work=dict(file='requester_work.json',id=saved['work']['artifact']['artifact_id']),
        prior_requester_review=dict(file='requester_review.json',
                                    id=saved['review']['result']['decision_id']))
    return result,refs

def history_publisher(request,state_root,folder,counts):
    saved, readback = effect.reconcile(request,state_root)
    if request['operation'] == 'CONFIRM_MOCK_BOOKING':
        d.need(saved['request'] == request,'SAME_KEY_CHANGED_PAYLOAD')
        counts['receipt_reuse'] += 1
    else:
        counts['pure_verification'] += 1
    n.save(folder/'receipt.json',saved['receipt'])
    n.save(folder/'readback.json',readback)
    n.save(folder/'history.json',dict(classification='SAVED_NATIVE_RECEIPT_CURRENT_LOCAL_READBACK',
        original_request=saved['request'],logical_key=saved['logical_key'],
        payload_sha256=saved['payload_sha256'],dispatch=False))
    source_path = state_root/'confirmed_source'/(effect.scope_key(request)+'.json')
    d.need(source_path.exists(),'UNKNOWN_REQUIRES_READBACK','CURRENT_STATUS_UNKNOWN')
    original = json.loads(source_path.read_text())
    body = original['body']
    observed_status = original.get('status',{}).get('value',dict(
        checked_at=original['entry']['effective_at'],valid_until=original['entry']['effective_at']+60))
    n.save(folder/'offer_body.json',body)
    n.save(folder/'source_history.json',original)
    result = d.report(request,'VERIFIED' if request['operation'] == 'VERIFY_EXISTING'
                      else 'MOCK_BOOKED',saved['slots'],body['offer']['total_minor'],
        source.projection(body,original['entry'],observed_status['checked_at'],
                          observed_status['valid_until']),
        saved['receipt']['artifact_id'])
    return result,dict(receipt=dict(file='receipt.json',id=saved['receipt']['artifact_id']),
        readback=dict(file='readback.json',id='snapshot:'+c.sha(readback)),
        history=dict(file='history.json',id=saved['logical_key']))

def publisher_contexts(events, policy):
    contexts = {}
    for event in events:
        request = event['request']
        key = p.task_ref(request)
        context = contexts.setdefault(key,dict(requester=policy['peer_root'],
            object_id=p.object_ref(request),purpose=request['operation'],
            uses=['LOCAL_CONTEXT'],source_revisions=[]))
        revision = event['inventory'].get('revision')
        if type(revision) is int and revision > 0 and revision not in context['source_revisions']:
            context['source_revisions'].append(revision)
    for context in contexts.values():
        context['source_revisions'].sort()
    return contexts

def run_peer_v01(*, role: str, input_data: dict, state_root: pathlib.Path,
                 output_root: pathlib.Path, channel: PipeChannel,
                 bootstrap: PipeChannel) -> dict | None:
    d.need(role in p.ROOTS,'PEER_ROLE')
    d.closed(input_data,('version','role','events','policy'))
    d.need(input_data['version'] == 'g54.peer_input.v01' and input_data['role'] == role,
           'PEER_INPUT_PROFILE')
    policy = input_data['policy']
    p.validate_policy(policy,role)
    events = input_data['events']
    d.need(type(events) is list and len(events) <= 8,'EVENT_BOUND')
    for event in events:
        d.safe_event(event['event_id'])
        # Identity fields are mandatory even for CLARIFY so the finite control
        # exchange can bind its event without guessing a request identity.
        request = event['request']
        for key in ('request_id','team_ref','venue_ref'):
            d.ident(request[key])
        d.need(type(request['revision']) is int and request['revision'] > 0 and
               request['operation'] in d.OPS,'EVENT_CONTROL_IDENTITY')
    d.need(len({x['event_id'] for x in events}) == len(events),'DUPLICATE_EVENT_ID')
    state_root.mkdir(parents=True,exist_ok=True)
    output_root.mkdir(parents=True,exist_ok=True)
    # Private signer capability never leaves this process and is never serialized.
    cap = c.crypto.generate_root_signer_capability_v01(root_id=policy['local_root'])
    keys = c.crypto.build_trusted_root_key_set_v01(capabilities=(cap,))
    public = c.crypto.trusted_root_key_set_to_plain_dict_v01(keys)
    bootstrap.send('POINTER',public)
    message = bootstrap.receive()
    d.need(message['kind'] == 'POINTER','BOOTSTRAP_PUBLIC_KEYSET')
    peer_keys = message['value']
    c.keyset(peer_keys)
    d.need(peer_keys['root_ids'] == [policy['peer_root']] and
           len(peer_keys['key_ids']) == 1,'BOOTSTRAP_EXPECTED_PEER_ROOT')
    e.immutable(output_root/'public_setup.json',dict(own_public=public,peer_public=peer_keys,
        provenance='PREASSIGNED_OPERATOR_BOOTSTRAP_CHANNEL',creates_permission=False))
    e.immutable(output_root/'operator_policy.json',policy)
    totals = new_counts()
    publisher_budget = None
    if role == 'publisher':
        contexts = publisher_contexts(events,policy)
        publisher_budget = e.PublisherBudget(state_root/'publisher_budget.json',contexts)
        e.immutable(output_root/'operator_contexts.json',contexts)
    seen = set()
    completed = []
    for event in events:
        event_id = event['event_id']
        folder = output_root/event_id
        folder.mkdir(parents=True,exist_ok=False)
        counts = new_counts()
        wire_before = channel.counters()
        bootstrap_before = bootstrap.counters()
        request = event['request']
        error = None
        historical = request['operation'] == 'VERIFY_EXISTING'
        try:
            request = p.event_request(event,role,folder)
            if request['operation'] == 'CONFIRM_MOCK_BOOKING':
                prior_path = state_root/('bookings' if role == 'publisher' else 'accepted')/(effect.scope_key(request)+'.json')
                if prior_path.exists():
                    prior = json.loads(prior_path.read_text())
                    historical = role == 'publisher' or prior['body']['booking'] is not None
            if historical:
                d.parse_request(request)
            else:
                p.check_history(request,policy,state_root,event_id)
            if request['operation'] == 'CLARIFY':
                raise d.Refusal('EXPLICIT_CLARIFY','CLARIFY')
        except d.Refusal as exc:
            error = exc
        try:
            if role == 'requester':
                local = p.local_policy(policy,request,event_id,peer_keys)
                budget = e.Budget(state_root/'budgets'/(c.sha(p.task_ref(request))+'.json'),
                                  p.task_ref(request))
                accepted_path = state_root/'accepted'/(effect.scope_key(request)+'.json')
                try:
                    if error is not None:
                        raise error
                    duplicate = (request['operation'] == 'CONFIRM_MOCK_BOOKING' and
                                 accepted_path.exists() and
                                 json.loads(accepted_path.read_text())['body']['booking'] is not None)
                    if historical or duplicate:
                        historical = True
                        result,refs = history_requester(request,state_root,folder,counts)
                    else:
                        d.need(policy['accept_offers'],'LOCAL_OFFER_POLICY')
                        if request['operation'] == 'CONFIRM_MOCK_BOOKING':
                            d.need(accepted_path.exists(),'CURRENT_SELECTED_OFFER_MISSING','NEEDS_CONFIRMATION')
                            selected = json.loads(accepted_path.read_text())
                            d.need(selected['body']['offer']['status'] == 'OFFER_READY' and
                                   d.constraints(selected['request']) == d.constraints(request),
                                   'SELECTED_OFFER_BINDING','NEEDS_CONFIRMATION')
                        result,refs,body = transport.retrieve(channel,cap,keys,local,
                            request,state_root,folder,counts)
                        n.save(accepted_path,dict(request=request,result=result,body=body,
                            work=json.loads((folder/'requester_work.json').read_text()),
                            review=json.loads((folder/'requester_review.json').read_text()),
                            authenticated=dict(peer_keys=peer_keys,
                                descriptor=json.loads((folder/'descriptor.json').read_text()),
                                status=json.loads((folder/'status.json').read_text()),
                                bundle=json.loads((folder/'bundle.json').read_text()),
                                request=json.loads((folder/'fetch_request.json').read_text()))))
                except d.Refusal as exc:
                    result,refs = report_failure(request,exc,role,folder,counts,historical)
                finally:
                    transport.close(channel,budget,request,local,cap,keys,folder,counts,event_id)
            else:
                def prepare():
                    return source.build(event_id,request,event['inventory'],policy,
                                        cap,keys,state_root,folder,counts)
                current,error,ops = transport.serve_event(channel,cap,keys,peer_keys,
                    request,event_id,policy,publisher_budget,state_root,folder,counts,seen,
                    prepare,error)
                if error is not None:
                    result,refs = report_failure(request,error,role,folder,counts,historical)
                elif current is not None:
                    body = current['body']
                    status = body['offer']['status']
                    receipt = None
                    if body['booking'] is not None:
                        status,receipt = 'MOCK_BOOKED',body['booking']['receipt_ref']
                        n.save(state_root/'confirmed_source'/(effect.scope_key(request)+'.json'),current)
                    observed_status = current.get('status',{}).get('value',dict(
                        checked_at=current['entry']['effective_at'],
                        valid_until=current['entry']['effective_at']+60))
                    result = d.report(request,status,body['offer']['slots'],body['offer']['total_minor'],
                        source.projection(body,current['entry'],observed_status['checked_at'],
                                          observed_status['valid_until']),receipt,
                        counts['search'],counts['dispatch'],counts['mutations'])
                    refs = dict(venue_work=dict(file='venue_work.json',id=body['source_work_ref']),
                        source_review=dict(file='source_review.json',id=body['source_review_ref']),
                        offer=dict(file='offer_body.json',id=body['source_record_ref']))
                    if receipt is not None:
                        refs['receipt'] = dict(file='receipt.json',id=receipt)
                elif historical or (request['operation'] == 'CONFIRM_MOCK_BOOKING' and
                        (state_root/'bookings'/(effect.scope_key(request)+'.json')).exists()):
                    try:
                        result,refs = history_publisher(request,state_root,folder,counts)
                    except d.Refusal as exc:
                        result,refs = report_failure(request,exc,role,folder,counts,True)
                else:
                    result,refs = report_failure(request,d.Refusal('PEER_CLOSED_WITHOUT_RETRIEVAL'),
                                                 role,folder,counts)
            n.save(folder/'report.json',dict(result,native_refs=refs))
            completed.append(dict(event_id=event_id,status=result['status']))
        except Exception as exc:
            # Preserve unknown native/I/O failures, then let the independent parent
            # observe the failed worker. Never relabel setup/teardown as success.
            n.save(folder/'unexpected_failure.json',dict(exception=type(exc).__name__,
                reason=str(exc),phase='PEER_EVENT',counts=counts,
                outcome='UNKNOWN_REQUIRES_READBACK' if counts['dispatch'] or counts['mutations']
                        else 'FAILED_BEFORE_DISPATCH',retry_dispatch=False))
            raise
        finally:
            n.save(folder/'counts.json',dict(local_observations=counts,
                wire=difference(channel.counters(),wire_before),
                bootstrap=difference(bootstrap.counters(),bootstrap_before),
                independent_observer_required=True))
            for key in totals:
                totals[key] += counts[key]
    n.save(output_root/'completion.json',dict(role=role,events=completed,
        counts=totals,wire=channel.counters(),bootstrap=bootstrap.counters(),
        classification='CANDIDATE_COMPLETION_ONLY'))
    return dict(role=role,events_processed=len(completed))
