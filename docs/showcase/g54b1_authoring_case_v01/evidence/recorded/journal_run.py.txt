"""Record a task-owned trusted command; no author-facing dispatch."""
import datetime as dt
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys
import time

WORK=Path(__file__).resolve().parent
label=sys.argv[1];argv=sys.argv[2:]
folder=WORK/'SUPERVISOR_EVIDENCE/commands'/label;folder.mkdir(parents=True,exist_ok=False)
receipt=dict(argv=argv,cwd=str(WORK),started=dt.datetime.now(dt.timezone.utc).isoformat(),supervisor_pid=os.getpid(),
             source_pins={p.name:hashlib.sha256(p.read_bytes()).hexdigest() for p in WORK.glob('*.py')})
start=time.monotonic()
with (folder/'stdout').open('wb') as out,(folder/'stderr').open('wb') as err:
    child=subprocess.Popen(argv,cwd=WORK,stdout=out,stderr=err,env=dict(os.environ,PYTHONDONTWRITEBYTECODE='1'))
    receipt['pid']=child.pid
    (folder/'started.json').write_text(json.dumps(receipt,indent=2)+'\n')
    print('START',label,'PID',child.pid,flush=True)
    while child.poll() is None:
        with (folder/'heartbeats.jsonl').open('a') as f:
            f.write(json.dumps(dict(time=dt.datetime.now(dt.timezone.utc).isoformat(),pid=child.pid,elapsed=time.monotonic()-start))+'\n')
        try:child.wait(timeout=15)
        except subprocess.TimeoutExpired:pass
    child.wait()
receipt.update(rc=child.returncode,reaped=True,finished=dt.datetime.now(dt.timezone.utc).isoformat(),elapsed_seconds=time.monotonic()-start,
               streams={n:dict(bytes=(folder/n).stat().st_size,sha256=hashlib.sha256((folder/n).read_bytes()).hexdigest()) for n in ('stdout','stderr')})
(folder/'receipt.json').write_text(json.dumps(receipt,indent=2)+'\n')
print('END',label,'RC',child.returncode,'SECONDS',receipt['elapsed_seconds'],flush=True)
print((folder/'stdout').read_text(),flush=True)
print((folder/'stderr').read_text(),file=sys.stderr,flush=True)
raise SystemExit(child.returncode)
