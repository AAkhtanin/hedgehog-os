"""Local operator policy and request history, never foreign permission."""
from hedgehog.external_drs import gate5_contracts_v01 as c, gate5_exchange_v01 as e
from . import domain as d

POLICY_FIELDS = ('local_root','peer_root','venue_ref','accept_offers','publish_enabled',
    'release_enabled','audience','release_uses','max_source_age','ttl_seconds',
    'status','approvals','booking_enabled')
ROOTS = {'publisher':'VenueRoot','requester':'ActivePlanetRoot'}

def validate_policy(policy, role):
    d.closed(policy,POLICY_FIELDS)
    d.need(policy['local_root'] == ROOTS[role] and
           policy['peer_root'] == ROOTS['requester' if role == 'publisher' else 'publisher'],
           'OPERATOR_ROOT_BINDING')
    d.ident(policy['venue_ref'])
    for name in ('accept_offers','publish_enabled','release_enabled','booking_enabled'):
        d.need(type(policy[name]) is bool,'POLICY_BOOLEAN')
    for name in ('audience','release_uses'):
        c.sequence(policy[name],4)
    d.need(set(policy['audience']) <= {policy['peer_root']} and
           set(policy['release_uses']) <= {'LOCAL_CONTEXT'}, 'POLICY_SCOPE')
    for name in ('max_source_age','ttl_seconds'):
        d.need(type(policy[name]) is int and 1 <= policy[name] <= 300,'TIME_POLICY')
    d.need(policy['status'] in ('ACTIVE','REVOKED','SUPERSEDED','UNAVAILABLE'),
           'STATUS_POLICY')
    d.need(type(policy['approvals']) is list and len(policy['approvals']) <= 16,
           'APPROVAL_BOUND')
    for approval in policy['approvals']:
        d.closed(approval,('approval_ref','scope','request_sha256'))
        d.ident(approval['approval_ref']); c.hash_value(approval['request_sha256'])
        d.need(approval['scope'] in ('SHIFT','BOOK'), 'APPROVAL_SCOPE')
    d.need(len({(a['approval_ref'],a['scope'],a['request_sha256'])
               for a in policy['approvals']}) == len(policy['approvals']),
           'DUPLICATE_APPROVAL')

def approved(policy, request, scope):
    return request['owner_approval_ref'] is not None and dict(
        approval_ref=request['owner_approval_ref'],scope=scope,
        request_sha256=c.sha(request)) in policy['approvals']

def event_request(event, role, folder):
    fields = {'event_id','request'} | ({'inventory'} if role == 'publisher' else set())
    optional = {'semantic_proposal','semantic_capture'}
    d.need(type(event) is dict and set(event) in (fields,fields|optional),'EVENT_FIELDS')
    d.safe_event(event['event_id'])
    request = event['request']
    if 'semantic_proposal' in event:
        proposal, capture = event['semantic_proposal'],event['semantic_capture']
        d.closed(proposal,('mode','parameters'))
        d.need(type(proposal['parameters']) is dict,'SEMANTIC_PARAMETERS')
        d.need('operation' not in proposal['parameters'],'SEMANTIC_DUPLICATE_MODE')
        proposed = dict(proposal['parameters'],operation=proposal['mode'])
        d.closed(capture,('version','origin','capture_ref','captured_at',
                          'request_sha256','proposal_sha256','original_request',
                          'original_response','provider','model'))
        d.need(capture['version'] == 'football.semantic_capture.v01' and
               capture['origin'] in ('CONTROLLED','LOCALLY_SUPPLIED_LIVE'),
               'CAPTURE_PROFILE')
        d.ident(capture['capture_ref']); d.uint(capture['captured_at'],'CAPTURE_TIME')
        d.need(capture['captured_at'] <= e.clock(),'CAPTURE_FUTURE')
        for name in ('provider','model'):
            d.ident(capture[name])
        d.need(capture['original_request'] == {'request':request} and
               capture['original_response'] == proposal and
               capture['request_sha256'] == c.sha(request) and
               capture['proposal_sha256'] == c.sha(proposal) and proposed == request,
               'SEMANTIC_CAPTURE_BINDING')
        e.immutable(folder/'semantic_capture.json',capture)
        e.immutable(folder/'semantic_consumption.json',dict(
            mode=proposal['mode'],parameters=proposal['parameters'],
            capture_ref=capture['capture_ref'],classification='UNTRUSTED_PROPOSAL'))
        request = proposed
    e.immutable(folder/'request.json',request)
    return request

def check_history(request, policy, state_root, event_id):
    sessions = d.parse_request(request)
    d.need(request['venue_ref'] == policy['venue_ref'], 'LOCAL_VENUE_POLICY')
    if request['operation'] == 'VERIFY_EXISTING':
        return sessions
    path = state_root/'requests'/(c.sha(request['request_id'])+'.json')
    history = c.decode(path.read_bytes()) if path.exists() else []
    # Persist received history even when validation subsequently refuses it.
    e.immutable(state_root/'events'/(event_id+'.json'),request)
    accepted = [x['request'] for x in history]
    if accepted:
        last_revision = max(x['revision'] for x in accepted)
        d.need(request['revision'] >= last_revision,'REQUEST_ROLLBACK')
        same = [x for x in accepted if x['revision'] == request['revision']]
        d.need(not same or all(d.constraints(x) == d.constraints(request) for x in same),
               'REQUEST_EQUIVOCATION')
    if any(s['shift'] for s in sessions):
        d.need(approved(policy,request,'SHIFT'),'SHIFT_NOT_OWNER_APPROVED',
               'NEEDS_CONFIRMATION')
        prior = [x for x in accepted if x['revision'] < request['revision'] and
                 all(s['allow_shift_minutes'] == 0 for s in x['sessions'])]
        d.need(bool(prior),'SHIFT_ORIGINAL_REQUEST_MISSING','NEEDS_CONFIRMATION')
        original = max(prior,key=lambda r:r['revision'])
        a, b = d.constraints(original),d.constraints(request)
        a = dict(a,revision=b['revision'],sessions=sorted(a['sessions'],key=lambda s:s['session_id']))
        restored = [dict(s,allow_shift_minutes=0) for s in b['sessions']]
        b = dict(b,sessions=sorted(restored,key=lambda s:s['session_id']))
        d.need(a == b,'SHIFT_CHANGED_HARD_CONDITIONS')
    if request['operation'] == 'CONFIRM_MOCK_BOOKING':
        d.need(policy['booking_enabled'] and approved(policy,request,'BOOK'),
               'BOOKING_CONSENT_REQUIRED','NEEDS_CONFIRMATION')
    if not any(x['request'] == request for x in history):
        history.append(dict(event_id=event_id,request=request))
        e.state_write(path,history)
    return sessions

def task_ref(request):
    # FIND and explicit BOOK consent are distinct operator-established purposes.
    # Revisions and event IDs never reset the allowances within either purpose.
    purpose = request['operation']
    return 'task:football:' + c.sha([request['team_ref'],request['request_id'],purpose])

def object_ref(request):
    return 'object:football:' + c.sha(request['venue_ref'])

def record_ref(event_id, request):
    return 'source:football:' + c.sha([event_id,request['team_ref'],request['request_id'],
                                    request['revision']])

def local_policy(operator, request, event_id, peer_keys):
    return dict(local_root=operator['local_root'],peer_root=operator['peer_root'],
        peer_keys=peer_keys,source_record_ref=record_ref(event_id,request),
        source_policy_ref='policy:football:release',object_id=object_ref(request),
        endpoint='endpoint:gate5:A',transport_scope='owner:football:inherited-pipes',
        domain='FOOTBALL',venue_ref=request['venue_ref'],team_ref=request['team_ref'],
        schemas=['FootballVenueOfferV01'],uses=['LOCAL_CONTEXT'],
        accept_offers=operator['accept_offers'],max_source_age=operator['max_source_age'],
        max_validity_horizon=300,request=request,operator_policy=operator,
        operator_policy_sha256=c.sha(operator))
